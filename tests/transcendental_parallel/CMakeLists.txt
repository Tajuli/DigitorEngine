cmake_minimum_required(VERSION 3.21)
project(DigitorTranscendentalParallelQualification LANGUAGES C CXX)
set(DIGITOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. digitor-engine-build)
add_executable(digitor_transcendental_parallel_test test_transcendental_parallel.cpp)
target_compile_features(digitor_transcendental_parallel_test PRIVATE cxx_std_20)
target_link_libraries(digitor_transcendental_parallel_test PRIVATE Digitor::Engine)
if(MSVC)
  target_compile_options(digitor_transcendental_parallel_test PRIVATE /W4 /WX)
else()
  target_compile_options(digitor_transcendental_parallel_test PRIVATE -Wall -Wextra -Werror)
endif()
