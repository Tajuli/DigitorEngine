cmake_minimum_required(VERSION 3.21)
project(DigitorNativeImageRuntimeQualification LANGUAGES C CXX)
set(DIGITOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. digitor-engine-build)
add_executable(digitor_native_image_runtime_test ../test_native_image_runtime.cpp)
target_compile_features(digitor_native_image_runtime_test PRIVATE cxx_std_20)
target_link_libraries(digitor_native_image_runtime_test PRIVATE Digitor::Engine)
if(MSVC)
  target_compile_options(digitor_native_image_runtime_test PRIVATE /W4 /WX)
else()
  target_compile_options(digitor_native_image_runtime_test PRIVATE -Wall -Wextra -Werror)
endif()
